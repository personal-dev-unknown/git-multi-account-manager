ALTER TABLE repositories ADD COLUMN initial_branch TEXT DEFAULT 'main';
ALTER TABLE repositories ADD COLUMN initialization_date TIMESTAMP;
ALTER TABLE repositories ADD COLUMN initial_commit_hash TEXT;
ALTER TABLE repositories ADD COLUMN auto_initialized BOOLEAN DEFAULT FALSE;
```

### Branches Table (NEW)
```sql
CREATE TABLE branches (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    repository_id INTEGER NOT NULL,
    name TEXT NOT NULL,
    
    -- Branch metadata
    is_default BOOLEAN DEFAULT FALSE,
    is_remote BOOLEAN DEFAULT FALSE,
    remote_name TEXT,  -- 'origin', 'upstream', etc.
    
    -- Tracking
    tracking_branch TEXT,  -- 'origin/main'
    last_commit_hash TEXT,
    last_commit_date TIMESTAMP,
    commits_ahead INTEGER DEFAULT 0,
    commits_behind INTEGER DEFAULT 0,
    
    -- Status
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (repository_id) REFERENCES repositories(id),
    UNIQUE(repository_id, name)
);
```

### Repository Initialization Log (NEW)
```sql
CREATE TABLE repository_initializations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    repository_id INTEGER NOT NULL,
    account_id INTEGER NOT NULL,
    
    -- Initialization details
    local_path TEXT NOT NULL,
    remote_created BOOLEAN DEFAULT FALSE,
    remote_url TEXT,
    initial_branch TEXT NOT NULL,
    initial_commit_hash TEXT,
    initial_commit_message TEXT,
    
    -- Statistics
    files_added INTEGER DEFAULT 0,
    lines_added INTEGER DEFAULT 0,
    
    -- Timestamps
    started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    
    FOREIGN KEY (repository_id) REFERENCES repositories(id),
    FOREIGN KEY (account_id) REFERENCES accounts(id)
);
```

## Implementation: Setup Repository Workflow

```python
class SetupRepositoryWorkflow:
    """
    Setup local project with remote Git repository
    """
    
    def __init__(self, db_manager, account_manager, api_clients):
        self.db = db_manager
        self.accounts = account_manager
        self.apis = api_clients
    
    def start(self):
        """Main entry point for setup workflow"""
        
        print("\n" + "="*60)
        print("Setup Local Project with Remote")
        print("="*60)
        
        # Step 0: Get directory
        directory = self.get_directory()
        if not directory:
            return
        
        # Step 1: Create remote repository
        remote_info = self.create_remote_repository()
        if not remote_info:
            return
        
        # Step 2: Initialize local Git
        if not self.initialize_git(directory, remote_info['account']):
            return
        
        # Step 3: Create branch
        branch_name = self.setup_branch(directory)
        if not branch_name:
            return
        
        # Step 4: Stage files (git add)
        if not self.stage_files(directory):
            return
        
        # Step 5: Create initial commit
        commit_hash = self.create_initial_commit(directory)
        if not commit_hash:
            return
        
        # Step 6: Push to remote
        if not self.push_to_remote(directory, remote_info, branch_name):
            return
        
        # Step 7: Save to database
        repo_id = self.save_to_database(
            directory, remote_info, branch_name, commit_hash
        )
        
        # Step 8: Show success
        self.show_success_summary(directory, remote_info, branch_name, repo_id)
    
    def get_directory(self) -> Optional[str]:
        """Get directory to setup"""
        
        current_dir = os.getcwd()
        
        print(f"\nCurrent directory: {current_dir}")
        print("\nWhat would you like to do?")
        print("[1] Setup this directory as a Git repository")
        print("[2] Setup a different directory")
        print("[3] Back to main menu")
        
        choice = input("\nSelect [1/2/3]: ").strip()
        
        if choice == "1":
            # Check if already a git repo
            if os.path.exists(os.path.join(current_dir, '.git')):
                print("\n⚠️  This directory is already a Git repository!")
                if not self.prompt_yes_no("Continue anyway?", default=False):
                    return None
            return current_dir
        
        elif choice == "2":
            custom_dir = input("\nEnter directory path: ").strip()
            custom_dir = os.path.expanduser(custom_dir)
            
            if not os.path.exists(custom_dir):
                print(f"❌ Directory does not exist: {custom_dir}")
                return None
            
            if os.path.exists(os.path.join(custom_dir, '.git')):
                print("\n⚠️  This directory is already a Git repository!")
                if not self.prompt_yes_no("Continue anyway?", default=False):
                    return None
            
            return custom_dir
        
        return None
    
    def create_remote_repository(self) -> Optional[dict]:
        """Create repository on remote platform"""
        
        print("\n" + "="*60)
        print("Step 1: Create Remote Repository")
        print("="*60)
        
        # Select platform
        print("\nSelect platform for remote repository:")
        print("[1] 🐙 GitHub")
        print("[2] 🦊 GitLab")
        print("[3] 🗃️  Bitbucket")
        print("[4] ← Back")
        
        platform_choice = input("\nSelect [1/2/3/4]: ").strip()
        
        platform_map = {
            "1": "github",
            "2": "gitlab",
            "3": "bitbucket"
        }
        
        platform = platform_map.get(platform_choice)
        if not platform:
            return None
        
        # Select account
        accounts = self.accounts.get_accounts_by_platform(platform)
        if not accounts:
            print(f"\n❌ No {platform} accounts configured")
            return None
        
        if len(accounts) == 1:
            account = accounts[0]
            print(f"\n✓ Using account: {account['username']}")
        else:
            print("\nSelect account:")
            for idx, acc in enumerate(accounts, 1):
                print(f"[{idx}] @{acc['username']} ({acc['email']})")
            
            choice = input(f"\nSelect [1-{len(accounts)}]: ").strip()
            try:
                account = accounts[int(choice) - 1]
            except (ValueError, IndexError):
                print("❌ Invalid selection")
                return None
        
        # Get repository details
        print("\n" + "="*60)
        print("Repository Details")
        print("="*60)
        
        # Default name from directory
        default_name = os.path.basename(os.getcwd())
        repo_name = input(f"\nRepository name [{default_name}]: ").strip() or default_name
        
        # Clean repo name (remove invalid characters)
        repo_name = re.sub(r'[^a-zA-Z0-9-_.]', '-', repo_name)
        
        print(f"\n  • Will be created as: {account['username']}/{repo_name}")
        print(f"  • URL will be: {platform}.com/{account['username']}/{repo_name}")
        
        description = input("\nDescription (optional): ").strip()
        
        # Visibility
        print("\nVisibility:")
        print("  [1] 🔒 Private (only you can see it)")
        print("  [2] 🌍 Public (anyone can see it)")
        
        vis_choice = input("\nSelect [1/2] (1): ").strip() or "1"
        is_private = vis_choice == "1"
        
        # Create repository via API
        print(f"\n⏳ Creating repository on {platform.title()}...")
        
        try:
            repo_data = self.apis.create_repository(
                platform=platform,
                account=account,
                name=repo_name,
                description=description,
                private=is_private
            )
            
            print(f"✓ Repository created: {platform}.com/{account['username']}/{repo_name}")
            print(f"  • Visibility: {'Private' if is_private else 'Public'}")
            
            return {
                'platform': platform,
                'account': account,
                'name': repo_name,
                'owner': account['username'],
                'description': description,
                'private': is_private,
                'ssh_url': repo_data['ssh_url'],
                'https_url': repo_data['https_url'],
                'web_url': repo_data['web_url']
            }
        
        except Exception as e:
            print(f"❌ Failed to create repository: {e}")
            return None
    
    def initialize_git(self, directory: str, account: dict) -> bool:
        """Initialize Git repository"""
        
        print("\n" + "="*60)
        print("Step 2: Initialize Local Git Repository")
        print("="*60)
        
        print(f"\nCurrent directory: {directory}")
        
        # Scan directory
        print("\n⏳ Scanning directory...")
        files = self.scan_directory(directory)
        
        print(f"Found files:")
        for category, file_list in files.items():
            print(f"  • {category}: {len(file_list)} files")
        
        print(f"  Total: {sum(len(f) for f in files.values())} files")
        
        # Initialize git
        print("\n⏳ Initializing Git repository...")
        
        try:
            result = subprocess.run(
                ['git', 'init'],
                cwd=directory,
                capture_output=True,
                text=True,
                check=True
            )
            
            print("✓ Initialized empty Git repository")
            
            # Configure git identity
            subprocess.run(
                ['git', 'config', 'user.name', account.get('name', account['username'])],
                cwd=directory,
                check=True
            )
            subprocess.run(
                ['git', 'config', 'user.email', account['email']],
                cwd=directory,
                check=True
            )
            
            print(f"\nGit configuration:")
            print(f"  user.name: {account.get('name', account['username'])}")
            print(f"  user.email: {account['email']}")
            
            return True
        
        except subprocess.CalledProcessError as e:
            print(f"❌ Failed to initialize Git: {e}")
            return False
    
    def setup_branch(self, directory: str) -> Optional[str]:
        """Create and configure default branch"""
        
        print("\n" + "="*60)
        print("Step 3: Configure Default Branch")
        print("="*60)
        
        print("\nChoose initial branch name:")
        print("[1] main (recommended, GitHub default)")
        print("[2] master (traditional)")
        print("[3] develop (development workflow)")
        print("[4] Custom branch name")
        
        choice = input("\nSelect [1/2/3/4] (1): ").strip() or "1"
        
        branch_map = {
            "1": "main",
            "2": "master",
            "3": "develop"
        }
        
        if choice in branch_map:
            branch_name = branch_map[choice]
        elif choice == "4":
            branch_name = input("\nEnter branch name: ").strip()
            if not branch_name:
                print("❌ Branch name cannot be empty")
                return None
            # Validate branch name
            if not re.match(r'^[a-zA-Z0-9/_-]+$', branch_name):
                print("❌ Invalid branch name")
                return None
        else:
            branch_name = "main"
        
        try:
            # Create branch (git will create it on first commit)
            # For now, just configure it
            subprocess.run(
                ['git', 'config', 'init.defaultBranch', branch_name],
                cwd=directory,
                check=True
            )
            
            print(f"\n✓ Configured branch: {branch_name}")
            print(f"✓ Set as default branch")
            
            return branch_name
        
        except subprocess.CalledProcessError as e:
            print(f"❌ Failed to configure branch: {e}")
            return None
    
    def stage_files(self, directory: str) -> bool:
        """Stage files for commit (git add)"""
        
        print("\n" + "="*60)
        print("Step 4: Stage Files for Commit")
        print("="*60)
        
        print("\n⏳ Scanning for files to add...")
        
        # Check for sensitive files
        sensitive_files = self.check_sensitive_files(directory)
        
        if sensitive_files:
            print("\n⚠️  WARNING: Sensitive files detected!")
            for file in sensitive_files:
                print(f"  • {file}")
            print("\nThese files should NOT be committed")
            
            if not self.prompt_yes_no("Continue anyway? (not recommended)", default=False):
                return False
        
        # Create .gitignore if needed
        gitignore_path = os.path.join(directory, '.gitignore')
        if not os.path.exists(gitignore_path):
            if self.prompt_yes_no("\nCreate .gitignore with recommended patterns?", default=True):
                self.create_gitignore(directory)
                print("✓ Created .gitignore")
        
        # Stage all files
        print("\n⏳ Staging files...")
        
        try:
            result = subprocess.run(
                ['git', 'add', '.'],
                cwd=directory,
                capture_output=True,
                text=True,
                check=True
            )
            
            # Get status
            status_result = subprocess.run(
                ['git', 'status', '--short'],
                cwd=directory,
                capture_output=True,
                text=True,
                check=True
            )
            
            staged_files = [line for line in status_result.stdout.split('\n') if line.strip()]
            
            print(f"✓ Staged {len(staged_files)} files for commit")
            
            # Show some files
            print("\nFiles to be committed:")
            for line in staged_files[:10]:
                print(f"  {line}")
            
            if len(staged_files) > 10:
                print(f"  ... and {len(staged_files) - 10} more files")
            
            return True
        
        except subprocess.CalledProcessError as e:
            print(f"❌ Failed to stage files: {e}")
            return False
    
    def create_initial_commit(self, directory: str) -> Optional[str]:
        """Create initial commit"""
        
        print("\n" + "="*60)
        print("Step 5: Create Initial Commit")
        print("="*60)
        
        # Get project name for default message
        project_name = os.path.basename(directory)
        default_message = f"Initial commit: {project_name}"
        
        print(f"\nDefault commit message:")
        print(f'"{default_message}"')
        
        print("\nOptions:")
        print("[1] Use default message")
        print("[2] Write custom message")
        
        choice = input("\nSelect [1/2] (1): ").strip() or "1"
        
        if choice == "2":
            commit_message = input("\nEnter commit message: ").strip()
            if not commit_message:
                commit_message = default_message
        else:
            commit_message = default_message
        
        print("\n⏳ Creating commit...")
        
        try:
            result = subprocess.run(
                ['git', 'commit', '-m', commit_message],
                cwd=directory,
                capture_output=True,
                text=True,
                check=True
            )
            
            # Get commit hash
            hash_result = subprocess.run(
                ['git', 'rev-parse', 'HEAD'],
                cwd=directory,
                capture_output=True,
                text=True,
                check=True
            )
            
            commit_hash = hash_result.stdout.strip()[:7]
            
            print("✓ Commit created")
            print(f"\nCommit details:")
            print(f"  Commit: {commit_hash}")
            print(f"  Message: {commit_message}")
            
            # Get stats
            stats = self.get_commit_stats(directory, commit_hash)
            print(f"  Files: {stats['files']} changed")
            print(f"  Insertions: {stats['insertions']} lines")
            
            return commit_hash
        
        except subprocess.CalledProcessError as e:
            print(f"❌ Failed to create commit: {e.stderr}")
            return None
    
    def push_to_remote(self, directory: str, remote_info: dict, branch_name: str) -> bool:
        """Push to remote repository"""
        
        print("\n" + "="*60)
        print("Step 6: Push to Remote Repository")
        print("="*60)
        
        account = remote_info['account']
        
        print(f"\nRemote repository: {remote_info['platform']}.com/{remote_info['owner']}/{remote_info['name']}")
        print(f"Branch: {branch_name} → origin/{branch_name}")
        
        # Select authentication method
        print("\nAuthentication method:")
        
        auth_options = []
        if account.get('ssh_key_path'):
            print("[1] 🔑 SSH (Recommended)")
            print(f"    Using: {account['ssh_key_path']}")
            auth_options.append(('1', 'ssh'))
        
        if account.get('pat_token'):
            idx = len(auth_options) + 1
            print(f"[{idx}] 🎫 HTTPS with PAT")
            auth_options.append((str(idx), 'https'))
        
        if not auth_options:
            print("❌ No authentication methods configured")
            return False
        
        choice = input(f"\nSelect [1-{len(auth_options)}] (1): ").strip() or "1"
        
        auth_method = dict(auth_options).get(choice, 'ssh')
        
        # Add remote
        try:
            if auth_method == 'ssh':
                remote_url = remote_info['ssh_url']
            else:
                remote_url = remote_info['https_url']
            
            subprocess.run(
                ['git', 'remote', 'add', 'origin', remote_url],
                cwd=directory,
                check=True,
                capture_output=True
            )
            
            print("\n⏳ Pushing to remote...")
            print(f"\ngit push -u origin {branch_name}\n")
            
            # Set up environment for SSH
            env = os.environ.copy()
            if auth_method == 'ssh' and account.get('ssh_key_path'):
                env['GIT_SSH_COMMAND'] = f"ssh -i {account['ssh_key_path']} -o IdentitiesOnly=yes"
            
            # Push (show output to user)
            result = subprocess.run(
                ['git', 'push', '-u', 'origin', branch_name],
                cwd=directory,
                env=env,
                capture_output=False,  # Show output directly
                text=True
            )
            
            if result.returncode == 0:
                print("\n✓ Push successful!")
                print(f"\nRemote tracking:")
                print(f"  • Local branch: {branch_name}")
                print(f"  • Remote branch: origin/{branch_name}")
                print(f"  • Tracking relationship: ✓ Configured")
                return True
            else:
                print("\n❌ Push failed")
                return False
        
        except subprocess.CalledProcessError as e:
            print(f"❌ Failed to push: {e}")
